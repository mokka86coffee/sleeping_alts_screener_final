# Патч · `detectors/flow_core.py` — характер спуска

1498 строк, поэтому патчем. Два блока.

Величина отвечает на вопрос «планомерное падение или обвал». Ровный
многобаровый съезд означает, что предложение отрабатывают порциями,
контролируемо; вертикаль одним движением — реакция на удар. Разворот
после первого говорит «работу закончили», после второго — только
«удар кончился», и это про другое.

Внутри модели раздела 0 это прямое различение следов: ступенчатый след
мм против разового выброса толпы. В коде такой величины нет ни одной —
`_read_vortex` сравнивает два последних экстремума и про путь между
ними не знает ничего.

**Величина только считается и отдаётся наружу.** Ни в один порог и ни
в один множитель не входит: наблюдённого разброса по рынку нет, а
порог, поставленный до замера, повторит историю churn.

### стало

```python
# ─────────────────────────────────────────────────────────────
# Характер спуска
# ─────────────────────────────────────────────────────────────
# Три величины на один нисходящий участок ряда. Три, а не одна
# сводная: они отвечают на разные вопросы и складывать их в индекс
# значит потерять именно то, ради чего они считаются.
#
#   bars        — длина участка в барах
#   monotony    — доля баров, идущих вниз, внутри участка
#   per_bar     — размах, делённый на длину, в долях |максимума|
#
# Планомерное падение: длинное, монотонное, с малой амплитудой на
# бар. Обвал: короткое, с большой амплитудой на бар, монотонность
# при этом тоже высокая — поэтому одной монотонности мало, и
# per_bar обязателен.
#
# Участок берётся последний: от глобального максимума хвоста до
# последующего минимума. Не «весь ряд вниз» — на реальном ряде
# спуск почти всегда с зазубринами, и требование строгого убывания
# нашло бы участок в два бара.
def descent(values: Sequence[float], tail: int = 40) -> dict[str, float]:
    """Характер последнего нисходящего участка ряда.

    Пустой словарь нулей, если участка нет: ряд короткий, идёт вверх
    или стоит. Нули означают «не измерено», и потребитель обязан
    это различать — отсутствие спуска не то же самое, что спуск
    длиной ноль.
    """
    empty = {"bars": 0.0, "monotony": 0.0, "per_bar": 0.0, "depth": 0.0}
    src = list(values[-tail:]) if tail and len(values) > tail else list(values)
    if len(src) < 4:
        return empty

    # Последний максимум, а не первый: max() при равенстве отдаёт
    # первый индекс, и плато перед обвалом целиком засчитывалось в
    # спуск. Проверка на [100]*20 + [70, 45, 22] давала 22 бара при
    # монотонности 0.14 — то есть обвал читался как долгий и рваный
    # съезд, ровно наоборот. Спуск начинается там, где началось
    # падение, а не там, где в последний раз стояла вершина.
    hi_idx = max(range(len(src)), key=lambda i: (src[i], i))
    if hi_idx >= len(src) - 2:
        # Максимум на самом краю: спуск ещё не начался либо длиной
        # в один бар. Мерить нечего.
        return empty

    seg = src[hi_idx:]
    lo_rel = min(range(len(seg)), key=lambda i: seg[i])
    if lo_rel < 2:
        return empty
    seg = seg[: lo_rel + 1]

    span = seg[0] - seg[-1]
    if span <= 0:
        return empty

    down = sum(1 for i in range(1, len(seg)) if seg[i] < seg[i - 1])
    steps = len(seg) - 1

    # Нормировка на модуль вершины, а не на размах: делить размах на
    # самого себя бессмысленно, а величины рядов несопоставимы между
    # монетами без приведения к их собственному масштабу.
    base = abs(seg[0]) if abs(seg[0]) > 1e-12 else 1.0

    return {
        "bars": float(steps),
        "monotony": down / steps if steps else 0.0,
        "per_bar": (span / steps) / base if steps else 0.0,
        "depth": span / base,
    }


# ─────────────────────────────────────────────────────────────
# Вортекс
# ─────────────────────────────────────────────────────────────
@dataclass
class VortexState:
```

## 2. Спуск дельты в `FlowState`

Первый ряд, к которому величина применяется, — кумулятивная дельта.
Она уже считается в `build_flow`, и её спуск отвечает ровно на
вопрос «продавец отрабатывает порциями или ударил разом».

### было

```python
    # Доля тейкерных покупок в обороте окна.
    buy_share: float = 0.5
```

### стало

```python
    # Доля тейкерных покупок в обороте окна.
    buy_share: float = 0.5

    # Характер последнего спуска кумулятивной дельты: длина в барах,
    # монотонность, амплитуда на бар, глубина. См. descent().
    #
    # Хранится словарём, а не четырьмя полями: величины читаются
    # только вместе, и по отдельности каждая обманывает. Высокая
    # монотонность есть и у планомерного съезда, и у обвала —
    # различает их per_bar.
    #
    # Ни в один порог не входит. Сначала замер разброса по рынку,
    # потом решение, что с ним делать.
    descent: dict[str, float] = field(default_factory=dict)
```

### было

```python
        delta_slope=_slope_of_flow(cum, avg_quote),
        buy_share=(buy / total) if total > 0 else 0.5,
    )
```

### стало

```python
        delta_slope=_slope_of_flow(cum, avg_quote),
        buy_share=(buy / total) if total > 0 else 0.5,

        # накопленный результат работы продавца, а ряд поинтервальных
        # дельт колеблется вокруг нуля, и «нисходящий участок» на нём
        # означает лишь несколько красных баров подряд.
```
