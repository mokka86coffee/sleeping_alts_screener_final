"""Сборка итоговой страницы отчёта."""

from __future__ import annotations

from core.models import Candidate, RunSnapshot
from render.blocks import (
    render_dashboard, render_header, render_legend, render_section,
)
from render.card import render_grid
from render.css import CSS
from render.funnel import render_funnel, render_sectors
from render.svg import shared_defs
from render.table import render_scan_table

FONTS_LINK = (
    '<link rel="preconnect" href="https://fonts.googleapis.com">'
    '<link rel="preconnect" href="https://fonts.gstatic.com" crossorigin>'
    '<link href="https://fonts.googleapis.com/css2?'
    'family=Inter:wght@200;300;400;500;700;800;900&display=swap" rel="stylesheet">'
)


def _bucket(candidates: list[Candidate], name: str) -> list[Candidate]:
    return [c for c in candidates if c.bucket == name]


def render_bucket_section(
    name: str,
    items: list[Candidate],
    desc: str,
    tier: int,
) -> str:
    """Секция с сеткой карточек. Пустая секция не выводится."""
    if not items:
        return ""
    return render_section(name, len(items), desc, tier) + render_grid(items)


def build_page(candidates: list[Candidate], snapshot: RunSnapshot) -> str:
    """Полный HTML отчёта."""
    viral = [c for c in candidates if c.is_viral]
    taiko = [c for c in candidates if c.taiko]
    dexe = [c for c in candidates if c.dexe]

    strong = _bucket(candidates, "strong")
    good = _bucket(candidates, "good")
    scout = _bucket(candidates, "scout")
    watch = _bucket(candidates, "watch")

    body_parts = [
        render_header(snapshot),
        render_dashboard(snapshot),
        render_legend(snapshot),
        render_funnel(snapshot),
        render_sectors(snapshot),
        render_scan_table(candidates),
        render_bucket_section(
            "VIRAL", viral,
            "спекулятивный разгон · размер позиции ¼, горизонт 1–3 дня", 1),
        render_bucket_section(
            "TAIKO", taiko,
            "подтверждённые развороты на старшем таймфрейме", 2),
        render_bucket_section(
            "DEXE", dexe,
            "отскок после дампа, post-pump", 2),
        render_bucket_section(
            "STRONG", strong,
            "совпало несколько независимых условий", 3),
        render_bucket_section(
            "GOOD", good,
            "сетап пригоден к торговле", 3),
        render_bucket_section(
            "SCOUT", scout,
            "ранняя стадия, вход преждевременен", 4),
        render_bucket_section(
            "WATCH", watch,
            "только наблюдение, действий нет", 4),
    ]

    body = "\n".join(p for p in body_parts if p)

    return f"""<!DOCTYPE html>
<html lang="ru">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width,initial-scale=1">
<title>Sleeping Alts Screener</title>
{FONTS_LINK}
<style>{CSS}</style>
</head>
<body>
{shared_defs()}
{body}
</body>
</html>
"""
